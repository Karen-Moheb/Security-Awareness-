def check_strength(password):

    score = 0

    if len(password) >= 8:
        score += 1

    if re.search(r"[A-Z]", password):
        score += 1

    if re.search(r"[a-z]", password):
        score += 1

    if re.search(r"[0-9]", password):
        score += 1

    if re.search(r"[@$!%*?&]", password):
        score += 1

    if score <= 2:
        return "Weak"

    elif score <= 4:
        return "Medium"

    else:
        return "Strong"

def init_db():

    with sqlite3.connect('logs.db') as conn:

        c = conn.cursor()

        c.execute("""

        CREATE TABLE IF NOT EXISTS logs (

            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user TEXT NOT NULL,
            password TEXT,
            action TEXT NOT NULL,
            details TEXT,
            time TEXT NOT NULL

        )

        """)

        conn.commit()

@app.route('/connect', methods=['POST'])
def connect():

    ip = request.remote_addr
    ua = request.headers.get('User-Agent')

    if 'iPhone' in ua:
        device = 'iPhone'

    elif 'Android' in ua:
        device = 'Android'

    elif 'Windows' in ua:
        device = 'Windows'

    else:
        device = 'Unknown'

    with sqlite3.connect('logs.db') as conn:

        c = conn.cursor()

        c.execute("""
        INSERT INTO logs(user, action, details, time)
        VALUES (?, ?, ?, ?)
        """, (

            device,
            "Connected To Open WiFi",
            f"IP: {ip}",
            datetime.now().isoformat()

        ))

        conn.commit()
    backup_logs()
    return render_template(
        "warning.html",
        device=device,
        ip=ip,
        time=datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    )
